cmake_minimum_required(VERSION 3.21)
set(BRIDGE_MODE host CACHE STRING "host or firmware")
set(SPRINGBOARD_ROM "" CACHE FILEPATH "Verified Palm test ROM; empty uses a passive test pattern")
option(SPRINGBOARD_ROM_SMOKE "Build the manually armed eight-read CS0 experiment" OFF)
set(SPRINGBOARD_ROM_MANIFEST "" CACHE FILEPATH "Independent verifier manifest for active ROM")
option(SPRINGBOARD_USB_DEBUG "Core-0-only USB CDC bring-up console" ON)
option(SPRINGBOARD_LATENCY_DIAGNOSTICS "Build separate bounded latency/constant diagnostic" OFF)
option(SPRINGBOARD_CONTINUOUS "Separate experimental continuous responder" OFF)
option(SPRINGBOARD_PIO_REARM "Separate PIO-rearmed continuous candidate" OFF)
option(SPRINGBOARD_REPLY_MARKER "PIO-rearm diagnostic: mark reply preload, not conditional enable" OFF)
option(SPRINGBOARD_PERSISTENT_REPLY "Separate persistent ARM responder experiment" OFF)
option(SPRINGBOARD_ASYNC_BOOT_AUTOARM "One-shot autoarm for module-installed Stage A" OFF)
option(SPRINGBOARD_ASYNC_STAGE_A "Separate disarmed Phase4 CS1 candidate" OFF)
option(SPRINGBOARD_BOOT_AUTO_ARM "Separate one-shot boot autoarm ROM candidate" OFF)
option(SPRINGBOARD_RX_FIRST "Use phased RX-first CPU service in the separate latency image" OFF)
option(SPRINGBOARD_DRIVE_MARKER "Diagnostic GP39 side-set data-drive marker" OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(DEPS "${CMAKE_CURRENT_LIST_DIR}/.deps")
if(NOT BRIDGE_MODE MATCHES "^(host|firmware)$")
    message(FATAL_ERROR "BRIDGE_MODE must be host or firmware")
endif()
if(BRIDGE_MODE STREQUAL "firmware")
    if(NOT PICO_BOARD STREQUAL "waveshare_core2350b" OR NOT PICO_PLATFORM STREQUAL "rp2350-arm-s")
        message(FATAL_ERROR "Use waveshare_core2350b / rp2350-arm-s")
    endif()
    if(NOT DEFINED PICO_SDK_PATH)
        set(PICO_SDK_PATH "${DEPS}/pico-sdk")
    endif()
    set(SDK_ARGS --sdk-path "${PICO_SDK_PATH}")
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/scripts/bootstrap.py"
    --mode "${BRIDGE_MODE}" --check ${SDK_ARGS} COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/scripts/check_pio_policy.py"
    COMMAND_ERROR_IS_FATAL ANY)
if(BRIDGE_MODE STREQUAL "firmware")
    include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
    project(rp2350_springboard C CXX ASM)
    include(cmake/picotool.cmake)
    pico_sdk_init()
    if(NOT PICO_RP2350 OR PICO_RP2350A)
        message(FATAL_ERROR "RP2350B required")
    endif()
else()
    project(rp2350_springboard_tests C)
    enable_testing()
endif()

    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/bootstrap.py"
        --mode "${BRIDGE_MODE}" --check ${SDK_ARGS}
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/check_pio_policy.py")

add_dependencies(sb_model validate_deps)
if(BRIDGE_MODE STREQUAL "host")
    file(GLOB EPIO_SOURCES CONFIGURE_DEPENDS "${DEPS}/epio/src/*.c")
    add_library(epio STATIC ${EPIO_SOURCES})
    set_target_properties(epio PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_include_directories(epio PUBLIC "${DEPS}/epio/include" "${DEPS}/apio/include")
    target_compile_definitions(epio PUBLIC APIO_EMULATION=1)
    target_compile_options(epio PUBLIC -fshort-enums)
    add_executable(test_springboard tests/test_springboard.c src/capture_program.c)
    target_link_libraries(test_springboard PRIVATE sb_model epio)
    target_compile_options(test_springboard PRIVATE -Wall -Wextra -Werror)
    add_test(NAME springboard COMMAND test_springboard)
    add_executable(test_rom_program tests/test_rom_program.c src/rom_program.c)
    target_link_libraries(test_rom_program PRIVATE sb_model epio)
    target_compile_options(test_rom_program PRIVATE -Wall -Wextra -Werror)
    add_test(NAME rom_program COMMAND test_rom_program)
    add_executable(test_rom_drive_marker tests/test_rom_program.c src/rom_program.c)
    target_link_libraries(test_rom_drive_marker PRIVATE sb_model epio)
    target_compile_definitions(test_rom_drive_marker PRIVATE SB_ROM_DRIVE_MARKER=1)
    target_compile_options(test_rom_drive_marker PRIVATE -Wall -Wextra -Werror)
    add_test(NAME rom_drive_marker COMMAND test_rom_drive_marker)
    add_executable(test_latency_program tests/test_latency_program.c src/latency_program.c src/rom_program.c)
    target_link_libraries(test_latency_program PRIVATE sb_model epio)
    target_compile_options(test_latency_program PRIVATE -Wall -Wextra -Werror)
    add_test(NAME latency_program COMMAND test_latency_program)
    add_executable(test_continuous_program tests/test_continuous_program.c src/continuous_program.c)
    target_link_libraries(test_continuous_program PRIVATE sb_model epio)
    target_compile_options(test_continuous_program PRIVATE -Wall -Wextra -Werror)
    add_test(NAME continuous_program COMMAND test_continuous_program)
    add_executable(test_pio_rearm tests/test_pio_rearm.c src/pio_rearm_program.c)
    target_link_libraries(test_pio_rearm PRIVATE sb_model epio)
    target_compile_options(test_pio_rearm PRIVATE -Wall -Wextra -Werror)
    add_test(NAME pio_rearm COMMAND test_pio_rearm)
    add_executable(test_pio_reply_marker tests/test_pio_rearm.c src/pio_rearm_program.c)
    target_link_libraries(test_pio_reply_marker PRIVATE sb_model epio)
    target_compile_definitions(test_pio_reply_marker PRIVATE SB_PIO_REPLY_MARKER=1)
    target_compile_options(test_pio_reply_marker PRIVATE -Wall -Wextra -Werror)
    add_test(NAME pio_reply_marker COMMAND test_pio_reply_marker)
    add_library(persistent_cosim SHARED tests/persistent_cosim.c src/pio_rearm_program.c)
    target_include_directories(persistent_cosim PRIVATE src)
    target_link_libraries(persistent_cosim PRIVATE epio)
    target_compile_definitions(persistent_cosim PRIVATE SB_PIO_REPLY_MARKER=1)
    target_compile_options(persistent_cosim PRIVATE -Wall -Wextra -Werror)
    add_executable(test_continuous_console tests/test_continuous_console.c src/continuous_console.c)
    target_include_directories(test_continuous_console PRIVATE src)
    target_compile_options(test_continuous_console PRIVATE -Wall -Wextra -Werror)
    add_test(NAME continuous_console COMMAND test_continuous_console)
    # Separate Phase1 native prototype; never included by any firmware target.
    add_executable(test_cs1_program tests/test_cs1_program.c src/cs1_program.c src/cs1_registers.c)
    target_link_libraries(test_cs1_program PRIVATE epio sb_model)
    target_include_directories(test_cs1_program PRIVATE src)
    target_compile_options(test_cs1_program PRIVATE -Wall -Wextra -Werror)
    add_test(NAME cs1_program COMMAND test_cs1_program)
    add_library(cs1_cosim SHARED tests/cs1_cosim.c src/cs1_program.c src/cs1_registers.c)
    target_link_libraries(cs1_cosim PRIVATE epio sb_model)
    target_include_directories(cs1_cosim PRIVATE src)
    target_compile_options(cs1_cosim PRIVATE -Wall -Wextra -Werror)
    add_executable(test_async_read_identity tests/test_async_read_identity.c src/cs1_program.c src/async_program.c)
    target_link_libraries(test_async_read_identity PRIVATE epio)
    target_compile_options(test_async_read_identity PRIVATE -Wall -Wextra -Werror)
    add_test(NAME async_read_identity COMMAND test_async_read_identity)
    add_executable(test_async_boot tests/test_async_boot.c src/async_boot.c)
    target_include_directories(test_async_boot PRIVATE src)
    target_compile_options(test_async_boot PRIVATE -Wall -Wextra -Werror)
    add_test(NAME async_boot COMMAND test_async_boot)
    add_library(async_cosim SHARED tests/async_cosim.c src/async_program.c src/cs1_registers.c src/async_stage_a.c)
    target_compile_definitions(async_cosim PRIVATE SB_ASYNC_CAPTURE=1)
    target_link_libraries(async_cosim PRIVATE epio sb_model)
    target_include_directories(async_cosim PRIVATE src include)
    target_compile_options(async_cosim PRIVATE -Wall -Wextra -Werror)
    add_executable(test_async_stage_a tests/test_async_stage_a.c src/async_stage_a.c src/cs1_registers.c)
    target_link_libraries(test_async_stage_a PRIVATE sb_model)
    target_include_directories(test_async_stage_a PRIVATE src include)
    target_compile_options(test_async_stage_a PRIVATE -Wall -Wextra -Werror)
    add_test(NAME async_stage_a COMMAND test_async_stage_a)
    # Optional standalone machine-code test artifact; never a flashable image.
    find_program(SB_ARM_GCC arm-none-eabi-gcc)
    find_program(SB_ARM_LD arm-none-eabi-ld)
    if(SB_ARM_GCC AND SB_ARM_LD)
        add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/cs1-reply.elf"
            COMMAND "${SB_ARM_GCC}" -mcpu=cortex-m33 -mthumb
                "-I${CMAKE_CURRENT_SOURCE_DIR}/src" -c
                "${CMAKE_CURRENT_SOURCE_DIR}/src/cs1_reply.S"
                -o "${CMAKE_CURRENT_BINARY_DIR}/cs1-reply.o"
            COMMAND "${SB_ARM_GCC}" -mcpu=cortex-m33 -mthumb -Wall -Wextra -Werror
                "-I${CMAKE_CURRENT_SOURCE_DIR}/src" -c
                "${CMAKE_CURRENT_SOURCE_DIR}/tests/cs1_abi.c"
                -o "${CMAKE_CURRENT_BINARY_DIR}/cs1-abi.o"
            COMMAND "${SB_ARM_LD}" --fatal-warnings -Ttext=0x20001000 -e sb_cs1_reply
                "${CMAKE_CURRENT_BINARY_DIR}/cs1-reply.o"
                -o "${CMAKE_CURRENT_BINARY_DIR}/cs1-reply.elf"
            DEPENDS src/cs1_reply.S src/cs1_engine.h src/cs1_registers.h
                tests/cs1_abi.c VERBATIM)
        add_custom_target(cs1_engine_elf DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/cs1-reply.elf")
        add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/async-reply.elf"
            COMMAND "${SB_ARM_GCC}" -mcpu=cortex-m33 -mthumb -c
                "${CMAKE_CURRENT_SOURCE_DIR}/src/async_reply.S" -o "${CMAKE_CURRENT_BINARY_DIR}/async-reply.o"
            COMMAND "${SB_ARM_LD}" --fatal-warnings -Ttext=0x20001000 -e sb_async_reply
                "${CMAKE_CURRENT_BINARY_DIR}/async-reply.o" -o "${CMAKE_CURRENT_BINARY_DIR}/async-reply.elf"
            DEPENDS src/async_reply.S VERBATIM)
        add_custom_target(async_engine_elf DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/async-reply.elf")
    endif()
else()
    set(ROM_ARGS)
    set(ROM_DEPENDS)
    if(SPRINGBOARD_ROM)
        set(ROM_ARGS --rom "${SPRINGBOARD_ROM}")
        set(ROM_DEPENDS "${SPRINGBOARD_ROM}")
    endif()
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/rom_image.h"
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/embed_rom.py"
            "${CMAKE_CURRENT_BINARY_DIR}/rom_image.h" ${ROM_ARGS}
        DEPENDS scripts/embed_rom.py ${ROM_DEPENDS} VERBATIM)
    add_custom_target(embed_rom_header DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/rom_image.h")
    add_executable(springboard_passive src/main.c src/capture_program.c)
    set(FIRMWARE_TARGETS springboard_passive)
    if(SPRINGBOARD_ROM_SMOKE OR SPRINGBOARD_LATENCY_DIAGNOSTICS OR SPRINGBOARD_CONTINUOUS OR SPRINGBOARD_PIO_REARM OR SPRINGBOARD_ASYNC_STAGE_A)
        if(NOT SPRINGBOARD_ROM OR NOT SPRINGBOARD_ROM_MANIFEST)
            message(FATAL_ERROR "ROM smoke build requires a verified ROM and its manifest")
        endif()
        add_custom_target(validate_active_rom
            COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/check_rom_model.py"
                "${SPRINGBOARD_ROM}" --manifest "${SPRINGBOARD_ROM_MANIFEST}"
            VERBATIM)
    endif()
    if(SPRINGBOARD_ROM_SMOKE)
        add_executable(springboard_rom_smoke src/rom_main.c src/rom_program.c)
        target_compile_definitions(springboard_rom_smoke PRIVATE
            SB_ROM_DRIVE_MARKER=$<BOOL:${SPRINGBOARD_DRIVE_MARKER}>)
        add_dependencies(springboard_rom_smoke validate_active_rom)
        list(APPEND FIRMWARE_TARGETS springboard_rom_smoke)
    endif()
    if(SPRINGBOARD_LATENCY_DIAGNOSTICS)
        if(NOT SPRINGBOARD_USB_DEBUG)
            message(FATAL_ERROR "Latency diagnostic requires USB for explicit mode selection and bounded arming")
        endif()
        add_executable(springboard_rom_latency src/latency_main.c src/latency_program.c)
        target_compile_definitions(springboard_rom_latency PRIVATE
            SB_LATENCY_RX_FIRST=$<BOOL:${SPRINGBOARD_RX_FIRST}>)
        if(SPRINGBOARD_RX_FIRST)
            target_sources(springboard_rom_latency PRIVATE src/cpu_reply.S)
            # An explicit assembly input enables GNU-stack checking for old
            # newlib objects too. The ELF requires no executable-stack request.
            target_link_options(springboard_rom_latency PRIVATE -Wl,-z,noexecstack)
        endif()
        add_dependencies(springboard_rom_latency validate_active_rom)
        list(APPEND FIRMWARE_TARGETS springboard_rom_latency)
    endif()
    if(SPRINGBOARD_CONTINUOUS)
        if(NOT SPRINGBOARD_USB_DEBUG)
            message(FATAL_ERROR "Continuous prototype requires explicit USB arm/stop")
        endif()
        add_executable(springboard_rom_continuous src/continuous_main.c src/continuous_program.c src/continuous_console.c src/cpu_reply.S)
        target_link_options(springboard_rom_continuous PRIVATE -Wl,-z,noexecstack)
        add_dependencies(springboard_rom_continuous validate_active_rom)
        list(APPEND FIRMWARE_TARGETS springboard_rom_continuous)
    endif()
    if(SPRINGBOARD_BOOT_AUTO_ARM AND (NOT SPRINGBOARD_PIO_REARM OR NOT SPRINGBOARD_PERSISTENT_REPLY))
        message(FATAL_ERROR "Boot autoarm requires the proven persistent PIO-rearm engine")
    endif()
    if(SPRINGBOARD_PIO_REARM)
        if(NOT SPRINGBOARD_USB_DEBUG)
            message(FATAL_ERROR "PIO rearm candidate requires explicit USB arm/stop")
        endif()
        if(SPRINGBOARD_PERSISTENT_REPLY)
            if(NOT SPRINGBOARD_REPLY_MARKER)
                message(FATAL_ERROR "Persistent experiment requires the preload marker")
            endif()
            set(REARM_TARGET springboard_rom_persistent)
            set(REARM_ARM_SOURCE src/persistent_reply.S)
        else()
            set(REARM_TARGET springboard_rom_pio_rearm)
            set(REARM_ARM_SOURCE src/cpu_reply.S)
        endif()
        set(REARM_MAIN src/pio_rearm_main.c)
        if(SPRINGBOARD_BOOT_AUTO_ARM)
            set(REARM_TARGET springboard_rom_boot_autoarm)
            set(REARM_MAIN src/boot_autoarm_main.c)
            set(REARM_EXTRA src/boot_autoarm_entry.S)
        endif()
        add_executable(${REARM_TARGET} ${REARM_MAIN} src/pio_rearm_program.c src/continuous_console.c ${REARM_ARM_SOURCE} ${REARM_EXTRA})
        target_compile_definitions(${REARM_TARGET} PRIVATE SB_PIO_REPLY_MARKER=$<BOOL:${SPRINGBOARD_REPLY_MARKER}>
            SB_PERSISTENT_REPLY=$<BOOL:${SPRINGBOARD_PERSISTENT_REPLY}>)
        target_link_options(${REARM_TARGET} PRIVATE -Wl,-z,noexecstack)
        add_dependencies(${REARM_TARGET} validate_active_rom)
        list(APPEND FIRMWARE_TARGETS ${REARM_TARGET})
    endif()
    if(SPRINGBOARD_ASYNC_STAGE_A)
        if(NOT SPRINGBOARD_USB_DEBUG OR SPRINGBOARD_BOOT_AUTO_ARM OR SPRINGBOARD_PIO_REARM)
            message(FATAL_ERROR "Stage A requires debug and a separate disarmed build")
        endif()
        set(ASYNC_ID_INPUT "${CMAKE_BUILD_TYPE}/${PICO_BOARD}/${PICO_PLATFORM}/${SPRINGBOARD_ASYNC_BOOT_AUTOARM}")
        foreach(source src/async_boot.c src/async_boot.h src/async_main.c src/async_reply.S src/async_reply.h src/async_program.c src/async_program.h src/async_stage_a.c src/async_stage_a.h src/cs1_registers.c src/cs1_registers.h src/bus_model.c src/bus_model.h include/visor_mailbox.h)
            file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${source}" digest)
            string(APPEND ASYNC_ID_INPUT "/${source}:${digest}")
        endforeach()
        file(SHA256 "${SPRINGBOARD_ROM}" digest)
        string(SHA256 ASYNC_BUILD_ID "${ASYNC_ID_INPUT}/rom:${digest}")
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/async_build.h" "#define SB_ASYNC_BUILD_ID \"${ASYNC_BUILD_ID}\"\n")
        add_executable(springboard_cs1_stage_a src/async_boot.c src/async_main.c src/async_reply.S
            src/async_stage_a.c src/async_program.c src/cs1_registers.c)
        target_include_directories(springboard_cs1_stage_a PRIVATE include)
        target_compile_definitions(springboard_cs1_stage_a PRIVATE SB_ASYNC_BOOT_AUTOARM=$<BOOL:${SPRINGBOARD_ASYNC_BOOT_AUTOARM}>)
        target_link_options(springboard_cs1_stage_a PRIVATE -Wl,-z,noexecstack)
        add_dependencies(springboard_cs1_stage_a validate_active_rom)
        list(APPEND FIRMWARE_TARGETS springboard_cs1_stage_a)

    foreach(target IN LISTS FIRMWARE_TARGETS)
        add_dependencies(${target} embed_rom_header)
        target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${DEPS}/apio/include")
        target_link_libraries(${target} PRIVATE sb_model pico_stdlib pico_multicore hardware_pio hardware_clocks)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
        pico_enable_stdio_uart(${target} 0)
        target_compile_definitions(${target} PRIVATE SB_USB_DEBUG=$<BOOL:${SPRINGBOARD_USB_DEBUG}>)
        pico_enable_stdio_usb(${target} ${SPRINGBOARD_USB_DEBUG})
        if(SPRINGBOARD_USB_DEBUG)
            # Never wait for a terminal or let debug I/O wait for USB buffer space.
            target_compile_definitions(${target} PRIVATE
                PICO_STDIO_USB_CONNECT_WAIT_TIMEOUT_MS=0
                PICO_STDIO_USB_STDOUT_TIMEOUT_US=0
                PICO_STDIO_DEADLOCK_TIMEOUT_MS=0
                PICO_ENABLE_USB_RESET_VIA_BAUD_RATE=0
                PICO_ENABLE_USB_RESET_VIA_VENDOR_INTERFACE=0)
        endif()
        # Run the complete image from SRAM; retain flash boot/UF2 loading.
        pico_set_binary_type(${target} copy_to_ram)
        pico_add_extra_outputs(${target})
    endforeach()
endif()
